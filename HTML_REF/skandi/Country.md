function money(amount, currency = settings.currency) {
    const locale =
      {
        EN: "en-US",
        SV: "sv-SE",
        NO: "nb-NO",
        DA: "da-DK",
        ES: "es-ES",
        FI: "fi-FI",
        DE: "de-DE",
        "FR-FR": "fr-FR",
        "FR-CA": "fr-CA",
        TH: "th-TH",
      }[settings.language] || "en-US";
    try {
      return new Intl.NumberFormat(locale, {
        style: "currency",
        currency,
        maximumFractionDigits: 0,
      }).format(Number(amount || 0));
    } catch (_) {
      return `${amount} ${currency}`;
    }
